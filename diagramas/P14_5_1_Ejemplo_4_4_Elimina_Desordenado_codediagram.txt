digraph P14_5_1_Ejemplo_4_4_Elimina_Desordenado_Flowchart {
    // Configuración general del grafo
    rankdir=TB;
    node [fontname="Helvetica", fontsize=10];
    edge [fontname="Helvetica", fontsize=9];

    // Definición de estilos de nodos
    node [shape=ellipse, style=filled, fillcolor=lightgrey]; // Inicio/Fin
    start [label="INICIO"];
    end [label="FIN"];

    node [shape=box, style=filled, fillcolor=lightblue]; // Proceso
    declare_vars [label="DECLARAR:\nScanner sc, int[] a, int n, int x,\nint i=0, boolean encontrado=false"];
    read_elements_loop [label="BUCLE for (i=0 hasta n-1)"];
    read_element [label="LEER a[i]"];
    find_loop_init [label="i = 0"];
    find_loop_proc [label="i++"];
    shift_loop [label="BUCLE for (k=i hasta n-1)"];
    shift_proc [label="a[k] = a[k+1]"];
    decrement_n [label="n--"];
    print_loop [label="BUCLE for (k=0 hasta n-1)"];
    print_element [label="IMPRIMIR a[k]"];
    close_scanner [label="Cerrar Scanner"];

    node [shape=parallelogram, style=filled, fillcolor=lightgoldenrod1]; // Entrada/Salida
    prompt_n [label="IMPRIMIR \"Ingrese el tamaño del arreglo (N): \""];
    read_n [label="LEER n"];
    prompt_elements [label="IMPRIMIR \"Ingrese los elementos del arreglo:\""];
    prompt_x [label="IMPRIMIR \"Ingrese el elemento a eliminar (X): \""];
    read_x [label="LEER x"];
    print_success [label="IMPRIMIR \"Elemento eliminado...\""];
    print_failure [label="IMPRIMIR \"Elemento no encontrado...\""];

    node [shape=diamond, style=filled, fillcolor=khaki, width=2, height=1.2]; // Decisión
    find_loop_cond [label="i < n &&\n!encontrado"];
    find_check [label="a[i] == x"];
    decision_encontrado [label="encontrado"];

    // Conexiones (flujo)
    start -> declare_vars -> prompt_n -> read_n -> prompt_elements -> read_elements_loop -> read_element -> read_elements_loop;
    read_elements_loop -> prompt_x [label="Fin del bucle"];
    prompt_x -> read_x -> find_loop_init -> find_loop_cond;

    find_loop_cond -> find_check [label="Sí"];
    find_loop_cond -> decision_encontrado [label="No"];
    find_check -> find_loop_proc [label="No"];
    find_check -> decision_encontrado [label="Sí, encontrado=true"];
    find_loop_proc -> find_loop_cond;

    decision_encontrado -> print_success [label="Sí"];
    decision_encontrado -> print_failure [label="No"];
    print_success -> decrement_n -> shift_loop -> shift_proc -> shift_loop;
    shift_loop -> print_loop [label="Fin del bucle"];
    print_failure -> print_loop;
    
    print_loop -> print_element -> print_loop;
    print_loop -> close_scanner [label="Fin del bucle"];
    close_scanner -> end;
}
