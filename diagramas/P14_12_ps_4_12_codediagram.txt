digraph P14_12_ps_4_12_Flowchart {
    // Configuración general del grafo
    rankdir=TB;
    node [fontname="Helvetica", fontsize=10];
    edge [fontname="Helvetica", fontsize=9];

    // Definición de estilos de nodos
    node [shape=ellipse, style=filled, fillcolor=lightgrey]; // Inicio/Fin
    start [label="INICIO"];
    end [label="FIN"];

    node [shape=box, style=filled, fillcolor=lightblue]; // Proceso
    declare_vars [label="DECLARAR:\nprecios[], scanner, numVentas,\ndias[], modelos[], cantidades[],\nrecaudadoPorModelo[], recaudadoPorDia[]"];
    read_ventas_loop [label="BUCLE for (i=0 hasta numVentas-1)"];
    read_venta [label="LEER dias[i], modelos[i], cantidades[i]"];
    calc_modelo_loop [label="BUCLE for (i=0 hasta numVentas-1)"];
    calc_modelo_proc [label="recaudadoPorModelo[modelo] += cantidad * precios[modelo]"];
    calc_dia_loop [label="BUCLE for (i=0 hasta numVentas-1)"];
    calc_dia_proc [label="recaudadoPorDia[dia] += cantidad * precios[modelo]"];
    find_best_loop [label="BUCLE for (i=0 hasta 7)"];
    update_best [label="maxRecaudado = recaudadoPorModelo[i]\nmodeloMasVendido = i + 1"];
    print_modelo_loop [label="BUCLE for (i=0 hasta 7)"];
    print_dia_loop [label="BUCLE for (i=0 hasta 29)"];
    close_scanner [label="Cerrar Scanner"];

    node [shape=parallelogram, style=filled, fillcolor=lightgoldenrod1]; // Entrada/Salida
    prompt_numVentas [label="IMPRIMIR \"Introduce el número de registros de venta: \""];
    read_numVentas [label="LEER numVentas"];
    prompt_ventas [label="IMPRIMIR \"Introduce los datos de cada venta...\""];
    print_no_ventas [label="IMPRIMIR \"No hay ventas para procesar.\""];
    print_reporte_modelo [label="IMPRIMIR \"Total recaudado por modelo:\""];
    print_modelo [label="IMPRIMIR \"Modelo [i+1]: $%.2f\""];
    print_reporte_dia [label="IMPRIMIR \"Total recaudado por día:\""];
    print_dia [label="IMPRIMIR \"Día [i+1]: $%.2f\""];
    print_mejor_modelo [label="IMPRIMIR \"El modelo que más dinero produjo fue el modelo [modeloMasVendido]\""];

    node [shape=diamond, style=filled, fillcolor=khaki, width=2, height=1.2]; // Decisión
    decision_numVentas [label="numVentas > 0"];
    check_dia_gt_0 [label="recaudadoPorDia[i] > 0"];
    check_best [label="recaudadoPorModelo[i] > maxRecaudado"];

    // Conexiones (flujo)
    start -> declare_vars -> prompt_numVentas -> read_numVentas -> decision_numVentas;
    decision_numVentas -> print_no_ventas [label="No"];
    print_no_ventas -> close_scanner;

    decision_numVentas -> prompt_ventas [label="Sí"];
    prompt_ventas -> read_ventas_loop -> read_venta -> read_ventas_loop;
    read_ventas_loop -> calc_modelo_loop [label="Fin del bucle"];
    
    calc_modelo_loop -> calc_modelo_proc -> calc_modelo_loop;
    calc_modelo_loop -> calc_dia_loop [label="Fin del bucle"];
    calc_dia_loop -> calc_dia_proc -> calc_dia_loop;
    calc_dia_loop -> find_best_loop [label="Fin del bucle"];
    
    find_best_loop -> check_best;
    check_best -> update_best [label="Sí"];
    check_best -> find_best_loop [label="No"];
    update_best -> find_best_loop;
    find_best_loop -> print_reporte_modelo [label="Fin del bucle"];

    print_reporte_modelo -> print_modelo_loop -> print_modelo -> print_modelo_loop;
    print_modelo_loop -> print_reporte_dia [label="Fin del bucle"];
    print_reporte_dia -> print_dia_loop -> check_dia_gt_0;
    check_dia_gt_0 -> print_dia [label="Sí"];
    check_dia_gt_0 -> print_dia_loop [label="No"];
    print_dia -> print_dia_loop;
    print_dia_loop -> print_mejor_modelo [label="Fin del bucle"];
    print_mejor_modelo -> close_scanner -> end;
}
