digraph P14_5_4_Ejemplo_4_4_Insertar_Desordenado_Flowchart {
    // Configuración general del grafo
    rankdir=TB;
    node [fontname="Helvetica", fontsize=10];
    edge [fontname="Helvetica", fontsize=9];

    // Definición de estilos de nodos
    node [shape=ellipse, style=filled, fillcolor=lightgrey]; // Inicio/Fin
    start [label="INICIO"];
    end [label="FIN"];

    node [shape=box, style=filled, fillcolor=lightblue]; // Proceso
    declare_vars [label="DECLARAR:\nScanner sc, int[] a, int n=0,\nchar continuar='s'"];
    insert_proc [label="a[n] = y\nn++"];
    print_loop [label="BUCLE for (i=0 hasta n-1)"];
    print_element [label="IMPRIMIR a[i]"];
    close_scanner [label="Cerrar Scanner"];

    node [shape=parallelogram, style=filled, fillcolor=lightgoldenrod1]; // Entrada/Salida
    prompt_y [label="IMPRIMIR \"Ingrese el elemento a insertar: \""];
    read_y [label="LEER y"];
    print_success [label="IMPRIMIR \"Elemento insertado...\""];
    print_current_array [label="IMPRIMIR \"Arreglo actual: \""];
    prompt_continue [label="IMPRIMIR \"¿Desea insertar otro? (s/n)\""];
    read_continue [label="LEER continuar"];
    print_lleno [label="IMPRIMIR \"No hay más espacio...\""];
    print_finalizado [label="IMPRIMIR \"Proceso finalizado.\""];

    node [shape=diamond, style=filled, fillcolor=khaki, width=2.5, height=1.2]; // Decisión
    main_loop_cond [label="n < a.length &&\n(continuar == 's' || continuar == 'S')"];
    check_if_full [label="n < a.length"];

    // Conexiones (flujo)
    start -> declare_vars -> main_loop_cond;
    main_loop_cond -> prompt_y [label="Sí"];
    main_loop_cond -> check_if_full [label="No"];

    prompt_y -> read_y -> insert_proc -> print_success -> print_current_array -> print_loop -> print_element -> print_loop;
    print_loop -> prompt_continue [label="Fin del bucle"];
    prompt_continue -> read_continue -> main_loop_cond;

    check_if_full -> print_finalizado [label="No, lleno"];
    check_if_full -> print_lleno [label="Sí, no lleno"];
    print_lleno -> print_finalizado;
    print_finalizado -> close_scanner -> end;
}
