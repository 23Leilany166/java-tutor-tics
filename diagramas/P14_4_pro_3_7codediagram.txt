digraph P14_4_pro_3_7_Flowchart {
    // Configuración general del grafo
    rankdir=TB;
    node [fontname="Helvetica", fontsize=10];
    edge [fontname="Helvetica", fontsize=9];

    // Definición de estilos de nodos
    node [shape=ellipse, style=filled, fillcolor=lightgrey]; // Inicio/Fin
    start [label="INICIO"];
    end [label="FIN"];

    node [shape=box, style=filled, fillcolor=lightblue]; // Proceso
    declare_vars [label="DECLARAR:\nventasChicas, ventasMedianas, ventasGrandes (int)\ncantidadVentas (int)"];
    init_loop [label="INICIAR bucle for\n(i desde 0 hasta cantidadVentas-1)"];
    increment_chicas [label="ventasChicas++"];
    increment_medianas [label="ventasMedianas++"];
    increment_grandes [label="ventasGrandes++"];
    close_scanner [label="Cerrar Scanner"];

    // Entrada (Input)
    node [shape=parallelogram, style=filled, fillcolor=lightgoldenrod1];
    read_cantidad [label="LEER cantidadVentas"];
    read_monto [label="LEER monto"];

    // Salida (Output / Display)
    node [shape=document, style=filled, fillcolor=lightcyan];
    prompt_cantidad [label="IMPRIMIR \"Introduce la cantidad de ventas a procesar: \""];
    prompt_monto [label="IMPRIMIR \"Introduce el monto de la venta #[i+1]: \""];
    print_resumen [label="IMPRIMIR Resumen de ventas:\n- Chicas: [ventasChicas]\n- Medianas: [ventasMedianas]\n- Grandes: [ventasGrandes]"];

    node [shape=diamond, style=filled, fillcolor=khaki, width=2, height=1]; // Decisión
    decision_chica [label="monto <= 200"];
    decision_grande [label="monto >= 400"];

    // Conexiones (flujo)
    start -> declare_vars -> prompt_cantidad -> read_cantidad -> init_loop;
    init_loop -> prompt_monto -> read_monto -> decision_chica;

    decision_chica -> increment_chicas [label="Sí"];
    decision_chica -> decision_grande [label="No"];
    decision_grande -> increment_grandes [label="Sí"];
    decision_grande -> increment_medianas [label="No"];

    {increment_chicas, increment_medianas, increment_grandes} -> init_loop; // Fin del bucle
    init_loop -> print_resumen [label="Fin del bucle"];
    print_resumen -> close_scanner -> end;
}
