digraph P14_8_ps_1_7_Flowchart {
    // Configuración general del grafo
    rankdir=TB;
    node [fontname="Helvetica", fontsize=10];
    edge [fontname="Helvetica", fontsize=9];

    // Definición de estilos de nodos
    node [shape=ellipse, style=filled, fillcolor=lightgrey]; // Inicio/Fin
    start [label="INICIO"];
    end [label="FIN"];

    node [shape=box, style=filled, fillcolor=lightblue]; // Proceso
    declare_vars [label="DECLARAR:\nScanner scanner, double precioVehiculo"];
    calc_pagos [label="enganche = precioVehiculo * 0.35\nresto = precioVehiculo - enganche\nmensualidad = resto / 36.0"];
    close_scanner [label="Cerrar Scanner"];

    node [shape=parallelogram, style=filled, fillcolor=lightgoldenrod1]; // Entrada/Salida
    prompt_precio [label="IMPRIMIR \"Introduce el precio del vehículo: \""];
    read_precio [label="LEER precioVehiculo"];
    print_reporte [label="IMPRIMIR \"--- Plan de Pagos ---\"\n- Enganche (35%%): $%.2f\n- Pago mensual (36 meses): $%.2f"];
    print_error [label="IMPRIMIR \"El precio debe ser mayor a cero.\""];

    node [shape=diamond, style=filled, fillcolor=khaki, width=2, height=1]; // Decisión
    decision_precio [label="precioVehiculo > 0"];

    // Conexiones (flujo)
    start -> declare_vars;
    declare_vars -> prompt_precio;
    prompt_precio -> read_precio;
    read_precio -> decision_precio;

    decision_precio -> calc_pagos [label="Sí"];
    decision_precio -> print_error [label="No"];
    
    calc_pagos -> print_reporte;
    {print_reporte, print_error} -> close_scanner;
    close_scanner -> end;
}
