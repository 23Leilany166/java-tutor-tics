digraph P14_5_5_Ejemplo_4_4_Modifica_Desordenado_Flowchart {
    // Configuración general del grafo
    rankdir=TB;
    node [fontname="Helvetica", fontsize=10];
    edge [fontname="Helvetica", fontsize=9];

    // Definición de estilos de nodos
    node [shape=ellipse, style=filled, fillcolor=lightgrey]; // Inicio/Fin
    start [label="INICIO"];
    end [label="FIN"];

    node [shape=box, style=filled, fillcolor=lightblue]; // Proceso
    declare_vars [label="DECLARAR:\nScanner sc, int[] a, int n, int x, int y,\nboolean encontrado=false"];
    read_elements_loop [label="BUCLE for (i=0 hasta n-1)"];
    read_element [label="LEER a[i]"];
    find_loop [label="BUCLE for (i=0 hasta n-1 && !encontrado)"];
    modify_proc [label="a[i] = y\nencontrado = true"];
    print_loop [label="BUCLE for (i=0 hasta n-1)"];
    print_element [label="IMPRIMIR a[i]"];
    close_scanner [label="Cerrar Scanner"];

    node [shape=parallelogram, style=filled, fillcolor=lightgoldenrod1]; // Entrada/Salida
    prompt_n [label="IMPRIMIR \"Ingrese el tamaño del arreglo (N): \""];
    read_n [label="LEER n"];
    print_vacio [label="IMPRIMIR \"El arreglo está vacío.\""];
    prompt_elements [label="IMPRIMIR \"Ingrese los elementos del arreglo:\""];
    prompt_x [label="IMPRIMIR \"Ingrese el elemento a modificar (X): \""];
    read_x [label="LEER x"];
    prompt_y [label="IMPRIMIR \"Ingrese el nuevo valor (Y): \""];
    read_y [label="LEER y"];
    print_modificado [label="IMPRIMIR \"Arreglo modificado:\""];
    print_no_encontrado [label="IMPRIMIR \"El elemento [X] no está en el arreglo.\""];

    node [shape=diamond, style=filled, fillcolor=khaki, width=2, height=1.2]; // Decisión
    decision_n_gt_0 [label="n > 0"];
    find_check [label="a[i] == x"];
    decision_encontrado [label="encontrado"];

    // Conexiones (flujo)
    start -> declare_vars -> prompt_n -> read_n -> decision_n_gt_0;
    decision_n_gt_0 -> print_vacio [label="No"];
    print_vacio -> close_scanner;

    decision_n_gt_0 -> prompt_elements [label="Sí"];
    prompt_elements -> read_elements_loop -> read_element -> read_elements_loop;
    read_elements_loop -> prompt_x [label="Fin del bucle"];
    prompt_x -> read_x -> prompt_y -> read_y -> find_loop;
    
    find_loop -> find_check -> find_loop;
    find_check -> modify_proc [label="Sí"];
    find_loop -> decision_encontrado [label="Fin del bucle"];

    decision_encontrado -> print_modificado [label="Sí"];
    decision_encontrado -> print_no_encontrado [label="No"];
    print_modificado -> print_loop -> print_element -> print_loop;
    print_loop -> close_scanner [label="Fin del bucle"];
    print_no_encontrado -> close_scanner;
    close_scanner -> end;
}
