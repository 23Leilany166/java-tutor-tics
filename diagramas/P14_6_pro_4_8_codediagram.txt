digraph P14_6_pro_4_8_Flowchart {
    // Configuración general del grafo
    rankdir=TB;
    node [fontname="Helvetica", fontsize=10];
    edge [fontname="Helvetica", fontsize=9];

    // Definición de estilos de nodos
    node [shape=ellipse, style=filled, fillcolor=lightgrey]; // Inicio/Fin
    start [label="INICIO"];
    end [label="FIN"];

    node [shape=box, style=filled, fillcolor=lightblue]; // Proceso
    declare_vars [label="DECLARAR:\nScanner sc, int aprobados=0, reprobados=0,\nmayoresA8=0, double sumaCalificaciones=0, int n"];
    loop_init [label="BUCLE for (i=0 hasta n-1)"];
    update_suma [label="sumaCalificaciones += cal"];
    increment_aprobados [label="aprobados++"];
    increment_reprobados [label="reprobados++"];
    increment_mayoresA8 [label="mayoresA8++"];
    calc_finales [label="Calcular promedio, porcAprobados, porcReprobados"];
    close_scanner [label="Cerrar Scanner"];

    node [shape=parallelogram, style=filled, fillcolor=lightgoldenrod1]; // Entrada/Salida
    prompt_n [label="IMPRIMIR \"Introduce el número de alumnos: \""];
    read_n [label="LEER n"];
    prompt_cal [label="IMPRIMIR \"Introduce la calificación del alumno #[i+1]: \""];
    read_cal [label="LEER cal"];
    print_no_alumnos [label="IMPRIMIR \"No hay alumnos para procesar.\""];
    print_reporte [label="IMPRIMIR Reporte Final:\n- Promedio\n- Aprobados\n- Reprobados\n- % Aprobados\n- % Reprobados\n- > 8"];

    node [shape=diamond, style=filled, fillcolor=khaki, width=2, height=1.2]; // Decisión
    decision_n_gt_0 [label="n > 0"];
    decision_aprobado [label="cal > 6"];
    decision_mayorA8 [label="cal > 8"];

    // Conexiones (flujo)
    start -> declare_vars -> prompt_n -> read_n -> decision_n_gt_0;
    decision_n_gt_0 -> print_no_alumnos [label="No"];
    print_no_alumnos -> close_scanner;

    decision_n_gt_0 -> loop_init [label="Sí"];
    loop_init -> prompt_cal -> read_cal -> update_suma -> decision_aprobado;
    
    decision_aprobado -> increment_aprobados [label="Sí"];
    decision_aprobado -> increment_reprobados [label="No"];
    increment_aprobados -> decision_mayorA8;
    increment_reprobados -> loop_init; // Vuelve al bucle
    decision_mayorA8 -> increment_mayoresA8 [label="Sí"];
    decision_mayorA8 -> loop_init [label="No"];
    increment_mayoresA8 -> loop_init;

    loop_init -> calc_finales [label="Fin del bucle"];
    calc_finales -> print_reporte -> close_scanner -> end;
}
